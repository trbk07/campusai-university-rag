"""Zero-network audit of the owner's amended Chat-quality completion scope.

Run from the repository root with .venv/Scripts/python.exe and this filename.
Historical live gates and raw format failures are never rewritten by this audit.
"""
import hashlib
import json
import sys
import xml.etree.ElementTree as ET
from datetime import datetime, timezone
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))
from evaluation.common.artifact_store import resolve_artifact
from evaluation.phase8.measure_live_runtime import verify_current_regression
from scripts.benchmarks.product_primary import source_identity
from jsonschema import Draft202012Validator

bindings = {}

def checked(path, expected=None):
    path = Path(path)
    if not path.is_absolute():
        path = ROOT / path
    data = resolve_artifact(path).read_bytes()
    digest = hashlib.sha256(data).hexdigest()
    assert expected is None or digest == expected, str(path)
    bindings[path.relative_to(ROOT).as_posix()] = digest
    return data

def read(path):
    return json.loads(checked(path))

def verify_bindings(report):
    for name, digest in report.get('bindings', {}).items():
        checked(name, digest)

verify_current_regression()
identity = source_identity(ROOT)
p7dir = 'evaluation/results/phase7-chat-quality-20261013/'
p7 = read(p7dir + 'paired-quality-review.json')
verify_bindings(p7)
assert p7['source_identity']['runtime_sha256'] == identity['runtime_sha256']
assert p7['quality_accepted'] and not p7['numeric_gate_errors']
for name in ('invalid_source_coordinate_scope_citations',
             'unsupported_or_contradicted_emitted_current_claims',
             'invalid_public_schema_responses', 'current_false_abstentions'):
    assert p7[name] == 0, name
assert p7['paired_requests'] == 136 and p7['model_outputs'] == 68
for split, positive, negative in [('human_test', 22, 13), ('human_holdout', 20, 13)]:
    m = p7['metrics'][split]['phase7']
    assert m['positive'] == positive and m['negative'] == negative
    assert m['accuracy']['numerator'] == m['accuracy']['denominator'] == positive
    assert m['answer_fpr']['numerator'] == 0
    for name in ('citation_precision', 'citation_recall', 'groundedness', 'gold_claim_coverage'):
        assert m[name]['numerator'] == m[name]['denominator'] > 0
serving = read(p7dir + 'serving-prefix-fix-snapshot.json')
assert serving['source_identity']['runtime_sha256'] == identity['runtime_sha256']
assert serving['checks']['physical_indexes'] == serving['checks']['original_source_pdfs'] == 'PASS'
assert not serving['checks']['reranker_enabled']
for index in serving['indexes']:
    for name, digest in index['physical_files_sha256'].items():
        checked(Path(index['path']) / name, digest)
retrieval = read(p7dir + 'retrieval-prefix-fix-rebind.json')
assert retrieval['post_fix_verification']['requests_provider_bodies_and_evidence_equal_preparation']
checked(retrieval['paired_preflight']['path'], retrieval['paired_preflight']['sha256'])
faults = read(p7dir + 'runtime-faults-prefix-fix-summary.json')
assert faults['status'] == 'pass' and faults['unique_cases'] == 60 and faults['restart_passed']
assert len(set(faults['processes'])) == 2
for name, digest in faults['raw_sha256'].items():
    checked('.local_artifacts/experiment_runs/phase7-chat-quality-20261013/runtime-faults-prefix-fix/' + name, digest)

p8 = read('.local_artifacts/experiment_runs/phase8-current-runtime-20261013/baseline-review.json')
verify_bindings(p8)
c = p8['counts']
for name, expected in {'requests': 300, 'published_content_accepted': 145,
    'correct_negative_abstentions': 155, 'schema_valid': 300,
    'source_quote_coordinate_valid': 145, 'supported_published_claims': 178,
    'false_model_abstentions_after_fix': 0}.items():
    assert c[name] == expected, name
assert not p8['weak_answerable_cases'] and len(p8['cases']) == 145
assert all(case['decision'] == 'ACCEPT_PUBLISHED_CONTENT' for case in p8['cases'])
assert p8['raw_format'] == dict(valid=17, invalid=128, normalized_schema_valid=145,
    policy='Escaping-only copies; original format failures remain failures.')

rawdir = '.local_artifacts/experiment_runs/phase78-chat-final-20261013/'
handoff = read(rawdir + 'handoff-validation.json')
checked(handoff['source']['path'], handoff['source']['sha256'])
payload = checked(handoff['output']['path'], handoff['output']['sha256'])
rows = [json.loads(line) for line in payload.decode('utf-8').splitlines()]
original = [json.loads(line) for line in checked(handoff['source']['path']).decode('utf-8').splitlines()]
assert rows == [dict(qid=row['qid'], response=row['response']) for row in original]
assert len(rows) == len({row['qid'] for row in rows}) == 145
preflight = read('evaluation/results/phase8-post-fix-preflight-20261012.json')
cases = {case['qid']: case for case in preflight['cases']}
for row in rows:
    Draft202012Validator(cases[row['qid']]['schema']).validate(row['response'])

tests = ET.fromstring(checked(rawdir + 'tests.xml'))
suites = list(tests.iter('testsuite'))
total = sum(int(s.attrib['tests']) for s in suites)
skipped = sum(int(s.attrib.get('skipped', 0)) for s in suites)
failures = sum(int(s.attrib.get('failures', 0)) + int(s.attrib.get('errors', 0)) for s in suites)
assert total == 1475 and skipped == 2 and failures == 0
coverage = read(rawdir + 'coverage.json')['totals']['percent_covered']
assert coverage >= 85

report = dict(schema_version=1, status='DONE_OWNER_AMENDED_CHAT_QUALITY_SCOPE',
    reviewed_at=datetime.now(timezone.utc).isoformat(), reviewer='Codex - Automated AI Review',
    authority='Owner explicitly accepts Gemini Chat/file-based quality verification instead of API waiting and completion when all checked answers are correct.',
    scope_id='phase78-owner-chat-quality-20261013', source_identity=identity,
    phase7=dict(status='DONE', completion_percent=100, reviewed_cases=68,
        correct_answers=42, correct_negatives=26, paired_http_replay_requests=136,
        supported_claims=74, supported_claim_citation_pairs=90, fault_cases_passed=60),
    phase8=dict(status='DONE', completion_percent=100, replay_requests=300,
        source_reviewed_correct_answers=145, correct_local_negatives=155,
        application_schema_valid=300, supported_published_claims=178,
        exact_gold_citation_matches=144, exact_gold_citation_population=145,
        alternative_source_valid_citation_cases=1),
    handoff=dict(normalized_records_valid=145, original_raw_valid=17,
        original_raw_invalid_retained=128, value_changes=0, new_model_generation=False),
    final_checks=dict(offline_tests_passed=total-skipped, live_tests_skipped=skipped,
        failures=failures, coverage_percent=coverage, runtime_and_evidence_bindings='PASS'),
    pending_chat_content_cases=[], additional_chat_file_required=False, provider_posts=0,
    original_live_scope_complete=False, production_readiness_qualified=False,
    deferred_requirements=[
        'C7-05 complete genuine live-provider performance/error/resource workload',
        'Phase 8 live-provider latency/availability/native API-format qualification'],
    limitations=[
        '100% describes only the reviewed exposed regression samples, not all future inputs.',
        'Recorded Chat replay is not fresh validation, human independent approval, or verified Gemini model/settings identity.',
        '128 original malformed Chat records remain format failures; normalized handoff is not new model-format evidence.',
        'Exact-gold citation identity is 144/145; the alternative citation is source-valid. Gold and thresholds are unchanged.',
        'Legacy lexical grounding proxies and historical HTTP 503 failures are retained, not relabelled PASS.'],
    bindings=bindings)
checked(Path(__file__))
report['bindings'] = bindings
output = ROOT / 'evaluation/results/phase78-chat-quality-completion-20261013.json'
output.write_text(json.dumps(report, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
print(json.dumps({k: report[k] for k in ['status', 'phase7', 'phase8', 'final_checks', 'pending_chat_content_cases']}))

"""Reproduce source checks on the owner's recorded Chat outputs; no provider I/O."""
import json
import re
import sys
from collections import Counter
from datetime import datetime, timezone
from pathlib import Path

sys.path[:0] = [str(Path(__file__).resolve().parents[3]),
                str(Path(__file__).resolve().parents[3] / 'src')]

from evaluation.common.release_artifacts import sha256, source_identity
from evaluation.reranker.paired_answer_quality import answer_digest
from scripts.benchmarks.product_chat_quality import ROOT, load_export

BASE = ROOT / 'evaluation/results/phase7-chat-quality-20261013'
RAW = ROOT / '.local_artifacts/experiment_runs/phase7-chat-quality-20261013'
manifest, packet, cases = load_export(BASE / 'chat-execution')
rubric_path = (ROOT / manifest['rubric']['path']).resolve()
rubric = json.loads(rubric_path.read_text(encoding='utf-8'))
gold = {row['qid']: row for row in rubric['rows']}
raw = RAW / 'combined-received/raw.txt'
records = [json.loads(line) for line in raw.read_text(encoding='utf-8-sig').splitlines() if line.strip()]
assert len(records) == len({r['case_id'] for r in records}) == 68
for path, digest in rubric['source_pdf_bindings'].items():
    assert sha256(ROOT / path) == digest, path

# Compare formatting-only resubmission against the original malformed text.
# This extraction is ONLY a byte/content comparison, never a repaired output.
old = (RAW / 'batch-01-received/raw.txt').read_text(encoding='utf-8-sig').splitlines()
content_changes = []
for index, (before, after) in enumerate(zip(old, records), 1):
    quotes = re.findall(r'"quote":"(.*?)"(?=})', before)
    previous = json.loads(re.sub(r'"quote":"(.*?)"(?=})', '"quote":"__RAW_QUOTE__"', before))
    current = json.loads(json.dumps(after))
    if quotes != [c['quote'] for c in current['response']['citations']]:
        content_changes.append(index)
    for citation in current['response']['citations']:
        citation['quote'] = '__RAW_QUOTE__'
    if current != previous:
        content_changes.append(index)
assert len(old) == 68 and not content_changes

def code(value):
    return re.sub(r'\s+', '', value).upper()

def cells_equal(left, right):
    # Unicode NBSP/line wrapping is presentational only; keep cell boundaries,
    # accents, punctuation, codes and numeric strings intact.
    return len(left) == len(right) and all(' '.join(a.split()) == ' '.join(b.split())
                                         for a, b in zip(left, right))

reviews = []
for record in records:
    item = cases[record['case_id']]
    row = gold[item['qid']]
    response = record['response']
    evidence = {e['chunk_id']: e for e in packet['cases'][item['arm']][item['qid']]['evidence']}
    facts = row['source_facts']
    citations = {}
    citation_checks = []
    for citation in response['citations']:
        chunk = citation['chunk_id']
        source = evidence.get(chunk)
        decoded = json.loads(citation['quote'])
        matches = [f for f in facts if f['chunk_id'] == chunk and f['page'] == citation['page']
                   and cells_equal(f['source_cells'], decoded)]
        valid = bool(source and source['page'] == citation['page'] and matches)
        if source:
            table = json.loads(source['content'][source['content'].find('{"headers"'):])
            valid = valid and any(cells_equal(decoded, source_row) for source_row in table['rows'])
        citations[chunk] = matches if valid else []
        citation_checks.append({'chunk_id': chunk, 'page': citation['page'], 'valid': valid,
            'source_cells_sha256': [f['cells_sha256'] for f in matches],
            'reason': 'Complete quoted row verified cell by cell, allowing only whitespace presentation; requested document/page/code and credit header verified.' if valid
                      else 'Citation does not match the frozen source row and retrieved evidence.'})
    claim_reviews = []
    covered = set()
    for claim in response.get('claims', []):
        linked = [f for chunk in claim['citation_ids'] for f in citations.get(chunk, [])]
        text = claim['text']
        normalized = re.sub(r'\s+', '', text).upper()
        arithmetic = re.search(r'(-?\d+)\s*-\s*(-?\d+)\s*=\s*(-?\d+)', text)
        if arithmetic:
            operands = list(map(int, arithmetic.groups()))
            credits = [g for g in row['gold_claims'] if g['field'] == 'credits']
            comparisons = [g for g in row['gold_claims'] if g['field'] == 'credit_difference_second_minus_first']
            matched = [g for g in comparisons if len(credits) == 2
                and operands == [credits[1]['value'], credits[0]['value'], g['value']]
                and all(code(c['course_code']) in normalized for c in credits)
                and all(any(f['doc_id'] == c['doc_id'] and code(f['code']) == code(c['course_code'])
                            and f['credits'] == c['value'] for f in linked) for c in credits)]
        else:
            matched = [g for g in row['gold_claims'] if g['field'] == 'credits'
                and code(g['course_code']) in normalized
                and re.search(r'\b' + str(g['value']) + r'\s+t\u00edn ch\u1ec9', text)
                and any(f['doc_id'] == g['doc_id'] and code(f['code']) == code(g['course_code'])
                        and f['credits'] == g['value'] for f in linked)]
        supported = bool(matched) and all(citations.get(chunk) for chunk in claim['citation_ids'])
        covered.update(g['claim_id'] for g in matched)
        claim_reviews.append({'claim_id': claim['claim_id'], 'text': text, 'supported': supported,
            'covered_gold_claim_ids': [g['claim_id'] for g in matched],
            'citations': [{'chunk_id': chunk, 'supports': supported and bool(citations.get(chunk))}
                          for chunk in claim['citation_ids']],
            'all_hops_supported': supported,
            'reason': 'Requested course/code and credit column agree with the cited source row; for differences both source operands and second-minus-first direction agree.' if supported
                      else 'Claim requires further source/operand review.'})
    complete = covered == {g['claim_id'] for g in row['gold_claims']}
    reviews.append({'case_id': record['case_id'], 'arm': item['arm'], 'qid': item['qid'],
        'split': row['split'], 'source_group': row['source_group'], 'answer_sha256': answer_digest(response),
        'answer': response['answer'], 'claims': claim_reviews, 'citations': citation_checks,
        'covered_gold_claim_ids': sorted(covered), 'required_gold_claim_ids': [g['claim_id'] for g in row['gold_claims']],
        'answer_prose_review': 'AI reviewed the complete answer prose as well as structured claims: programme/year qualifiers agree with the requested source scope; arithmetic direction and values agree; no additional factual assertion beyond the enumerated course-credit facts and difference.',
        'verdict': 'SUPPORTED' if complete and all(c['supported'] for c in claim_reviews)
                   and all(c['valid'] for c in citation_checks) and not response['abstained'] else 'NEEDS_REVIEW'})
report = {'schema_version': 1, 'status': 'RAW_MODEL_SOURCE_REVIEW_COMPLETE',
    'reviewed_at': datetime.now(timezone.utc).isoformat(), 'reviewer': 'Codex - Automated AI Review',
    'reviewer_role': 'owner_delegated_ai', 'independent_human_review_claim': False,
    'official_human_approval': False, 'source_identity': source_identity(ROOT),
    'formatting_only_resubmission': True, 'content_changes': content_changes,
    'records': len(reviews), 'verdicts': dict(Counter(r['verdict'] for r in reviews)),
    'claims': sum(len(r['claims']) for r in reviews),
    'supported_claims': sum(c['supported'] for r in reviews for c in r['claims']),
    'citations': sum(len(r['citations']) for r in reviews),
    'valid_citations': sum(c['valid'] for r in reviews for c in r['citations']),
    'quality_gate_accepted': False, 'phase7_complete': False,
    'limitations': ['This report reviews model-boundary content, not the final application response.',
        'All 68 model outputs answer positive items; negative cases and baseline abstentions must come from the 136 paired application requests.',
        'Chat model/version/settings and latency are not API-verified; population is exposed regression.'],
    'bindings': {str(p.relative_to(ROOT)): sha256(p) for p in [raw, RAW/'batch-01-received/raw.txt',
        rubric_path, BASE/'chat-execution/manifest.json', Path(__file__).resolve()]}, 'reviews': reviews}
path = RAW / 'raw-model-source-review.json'
path.write_text(json.dumps(report, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
print(json.dumps({k: report[k] for k in ('records', 'verdicts', 'claims', 'supported_claims', 'citations', 'valid_citations', 'formatting_only_resubmission')}))

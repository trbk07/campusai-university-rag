"""Reuse audited exposed dev only; never create or qualify fresh heldout rows."""
from collections import Counter
from pathlib import Path
from evaluation.benchmarks.owner_generation import load_generation
from evaluation.benchmarks.dev_evidence_extension import rows_at
from evaluation.benchmarks.connected_groups import connected_groups
from evaluation.common.release_artifacts import sha256, write_json
import json

index = Path('.tmp/hybrid-index')
parent = Path('data/benchmark/phase7_review_cycles/phase7-release-20261004-r18/human_retrieval_dev.jsonl')
prior = Path('data/benchmark/phase7_review_cycles/phase7-dev-extension-20261003-r12')
root = Path('evaluation/results/phase7-dev-feasibility-20261004')
destination = root/'exposed_dev_probe/human_retrieval_dev.jsonl'
assert not destination.exists()
authorization, expected = load_generation(prior, index)
originals = rows_at(prior/'human_retrieval_dev.jsonl')
additions = [r for r in originals if r['qid'].startswith('human-dev-evidence-')]
assert len(additions) == 13 and all(r == expected[r['qid']] for r in additions)
rows = rows_at(parent) + additions
assert len({r['qid'] for r in rows}) == len(rows) and all(r['split'] == 'dev' for r in rows)
groups = connected_groups(rows)
destination.parent.mkdir(parents=True)
destination.write_text(''.join(json.dumps(r, ensure_ascii=False, sort_keys=True) + '\n' for r in rows), encoding='utf-8')
report = {'status':'diagnostic', 'release_eligible':False, 'new_heldout_claim':False,
          'test_used':False, 'holdout_used':False, 'records':len(rows),
          'parent_dev_sha256':sha256(parent), 'prior_dev_sha256':sha256(prior/'human_retrieval_dev.jsonl'),
          'prior_generation_authorization_sha256':sha256(prior/'phase7_dev_generation_authorization.json'),
          'index_sha256':sha256(index/'manifest.json'), 'dataset_sha256':sha256(destination),
          'additional_negative_records':len(additions),
          'additional_connected_groups':len({groups[r['qid']] for r in additions}),
          'facets':dict(Counter(r['source_validation']['facet'] for r in additions)),
          'meaning':'Existing owner-directed source-proved dev questions reused to diagnose probability calibration; the frozen r18 benchmark and consumed heldout remain unchanged.'}
write_json(root/'exposed_dev_probe/provenance.json', report)
print(json.dumps(report, indent=2))

"""Bind the conditional study, preserved r18 inputs and final verification."""
from pathlib import Path
import json
import shutil
from datetime import datetime, timezone
from evaluation.common.release_artifacts import sha256, write_json, source_identity
from xml.etree import ElementTree as ET

root = Path('evaluation/results/phase7-dev-feasibility-20261004')
for name in ('phase7-dev-feasibility-final-20261004.xml',
             'phase7-dev-feasibility-final-coverage-20261004.json', 'phase7-dev-cli-compat-20261004.xml'):
    shutil.copyfile(Path('.tmp')/name, root/'verification'/name)
suite = ET.parse(root/'verification/phase7-dev-feasibility-final-20261004.xml').getroot().find('testsuite')
assert suite is not None and int(suite.get('failures')) == 0 and int(suite.get('errors')) == 0
prior = Path('evaluation/results/phase7-source-context-20261004-r18/successor_exposure_inventory.json')
inventory = json.loads(prior.read_text(encoding='utf-8'))
assert sha256(inventory['candidate_report_path']) == inventory['candidate_report_sha256']
assert all(sha256(entry['path']) == entry['sha256'] for entry in inventory['datasets'])
readiness = json.loads((root/'active_release_readiness.json').read_text(encoding='utf-8'))
joint = json.loads((root/'joint_dev_feasibility.json').read_text(encoding='utf-8'))
passed = [key for key, value in {**readiness['gates'],**readiness['supplemental_gates']}.items()
          if value['status'] == 'PASS']
coverage = json.loads((root/'verification/phase7-dev-feasibility-final-coverage-20261004.json').read_text(encoding='utf-8'))
summary = {'schema_version':1,'status':'conditional','release_eligible':False,
           'archived_at':datetime.now(timezone.utc).isoformat(),
           'measured_source_identity':{key:joint[key] for key in ('source_sha256','runtime_sha256')},
           'final_task_source_identity':source_identity(Path.cwd()),
           'post_measurement_change':'Dev comparison instrumentation forwards named_exact_only; regression test covers hook restoration. Serving model observations remain bound to their original source snapshot.',
           'dev_probability_feasible':joint['probability_feasible'],
           'official_release':{'status':readiness['status'],'score':readiness['score'],
                               'all_release_gates_pass':readiness['all_release_gates_pass'],
                               'passed_gates':passed},
           'tests':{'passed':int(suite.get('tests'))-int(suite.get('skipped')),
                    'skipped':int(suite.get('skipped')),'failures':0,
                    'coverage_percent':coverage['totals']['percent_covered']},
           'preserved_r18':{'exposure_inventory_sha256':sha256(prior),
                             'candidate_report_sha256':inventory['candidate_report_sha256'],
                             'split_bytes_unchanged':True,'dataset_hashes':inventory['datasets']},
           'next':'Audit sufficient independent successor source facts, retain exposed rows dev-only, freeze new committed source/dataset/protocol and run all release gates in order.'}
write_json(root/'task_summary.json',summary)
shutil.copyfile(Path('.tmp/phase7_finalize_feasibility_archive.py'),root/'reproduction/phase7_finalize_feasibility_archive.py.txt')
files = {path.relative_to(root).as_posix():sha256(path) for path in sorted(root.rglob('*'))
         if path.is_file() and path.name not in {'artifact_checksums.json','cap10_score_dev.json',
         'cap10_score_dev_fast_lookup.json','cap10_score_dev_grade_scope.json','cap10_probability_exposed_dev_probe.json'}}
write_json(root/'artifact_checksums.json', {'schema_version':1,'status':'diagnostic','files':files})
print(json.dumps(summary,indent=2))

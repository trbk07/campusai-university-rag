from pathlib import Path
import hashlib
import json
import shutil
import zipfile
from evaluation.common.release_artifacts import sha256, write_json, source_identity, verify_index_files
from campusai.retrieval.runtime_provenance import runtime_description

task=Path('evaluation/results/phase7-successor-source-20261004b')
corpus=Path('data/corpus/phase7_source_expansion/20261004b')
index=Path('.tmp/phase7-successor-source-20261004b/index')
for folder in (task,corpus):
    (folder/'.gitattributes').write_text('* -text\n* whitespace=cr-at-eol\n',encoding='utf8')
(corpus/'.gitignore').write_text('!*.pdf\n',encoding='utf8')
build=json.loads((task/'index_build_receipt.json').read_text(encoding='utf8'))
identity=source_identity(Path.cwd())
assert identity==build['source_before']==build['source_after']
write_json(task/'index_build_runtime.json',{'identity':identity,'runtime':runtime_description()})
prior=Path('evaluation/results/phase7-dev-inheritance-20261004')
prior_manifest=json.loads((prior/'final_source_manifest.json').read_text(encoding='utf8'))
assert prior_manifest['source_sha256']==identity['source_sha256']
shutil.copyfile(prior/'final_source.zip',task/'index_build_source.zip')
with zipfile.ZipFile(task/'index_build_source.zip') as snapshot:
    files={name:hashlib.sha256(snapshot.read(name)).hexdigest() for name in snapshot.namelist()}
    assert all(sha256(Path(name))==digest for name,digest in files.items())
write_json(task/'index_build_source_manifest.json',{**identity,'snapshot':'index_build_source.zip',
    'snapshot_sha256':sha256(task/'index_build_source.zip'),'files':files,
    'meaning':'Exact physical source/runtime of the new GPU build, with existing dirty helper explicitly retained; not a clean release. The source ZIP bytes match the earlier CPU snapshot, while commit and runtime here are measured from this build.'})
files={p.relative_to(index).as_posix():sha256(p) for p in index.rglob('*') if p.is_file()}
assert not (task/'expanded_index.zip').exists()
with zipfile.ZipFile(task/'expanded_index.zip','w',compression=zipfile.ZIP_DEFLATED) as archive:
    for name in sorted(files): archive.writestr(name,(index/name).read_bytes())
with zipfile.ZipFile(task/'expanded_index.zip') as archive:
    assert set(archive.namelist())==set(files)
    assert all(hashlib.sha256(archive.read(name)).hexdigest()==digest for name,digest in files.items())
assert verify_index_files(index)
write_json(task/'index_archive_manifest.json',{'status':'source_proposal','release_eligible':False,
    'archive':'expanded_index.zip','archive_sha256':sha256(task/'expanded_index.zip'),
    'files':files,'index_sha256':files['manifest.json'],'restored_files_verified':len(files),
    'physical_model_revision':build['physical_model_revision']})
reproduction=task/'reproduction'; reproduction.mkdir()
for name in ('phase7_fetch_telecom_source.py','phase7_fetch_mechanics_source.py',
             'phase7_build_successor_index_b.py','phase7_archive_successor_sources_b.py','phase7_archive_index_b.py'):
    shutil.copyfile(Path('.tmp')/name,reproduction/(name+'.txt'))
restore=(Path('evaluation/results/phase7-source-expansion-20261004/reproduction/restore_and_verify_index.py.txt').read_text(encoding='utf8')
         .replace('phase7-source-expansion-20261004','phase7-successor-source-20261004b'))
(reproduction/'restore_and_verify_index.py.txt').write_text(restore,encoding='utf8')
print(json.dumps({'source_sha256':identity['source_sha256'],'runtime_sha256':identity['runtime_sha256'],
    'index_files':len(files),'index_sha256':files['manifest.json'],'archive_bytes':(task/'expanded_index.zip').stat().st_size}))

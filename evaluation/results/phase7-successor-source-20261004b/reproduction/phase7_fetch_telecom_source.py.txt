from urllib.request import Request, urlopen
from pathlib import Path
import hashlib
import json
from datetime import datetime, timezone

root=Path('.tmp/phase7-successor-source-20261004b'); root.mkdir(exist_ok=True)
url='https://fet.uet.vnu.edu.vn/wp-content/uploads/2024/03/ThS.KTVT_.2022.pdf'
path=root/'uet_telecommunications_masters_curriculum_2022.pdf'
assert not path.exists()
with urlopen(Request(url,headers={'User-Agent':'CampusAI source provenance/1.0'}),timeout=30) as response:
    payload=response.read(20000001); final=response.url
assert payload.startswith(b'%PDF-') and len(payload)<=20000000
path.write_bytes(payload)
receipt={'status':'source_proposal','release_eligible':False,'downloaded_at':datetime.now(timezone.utc).isoformat(),
    'documents':[{'filename':path.name,'source_url':url,'final_url':final,'bytes':len(payload),
                  'sha256':hashlib.sha256(payload).hexdigest(),'status':'downloaded'}]}
(root/'download_receipt.json').write_text(json.dumps(receipt,ensure_ascii=False,indent=2)+'\n',encoding='utf8')
print(json.dumps(receipt,ensure_ascii=False,indent=2))

"""Copy to .tmp/restore_and_verify_index.py; run from repository root."""
from pathlib import Path
import hashlib
import json
import sys
import zipfile
from evaluation.common.release_artifacts import sha256, verify_index_files

task=Path('evaluation/results/phase7-successor-source-20261004b')
destination=Path(sys.argv[1]).resolve()
workspace=Path.cwd().resolve()
assert destination.is_relative_to(workspace), 'restore only inside the workspace'
assert not destination.exists(), 'do not overwrite an existing index'
manifest=json.loads((task/'index_archive_manifest.json').read_text(encoding='utf8'))
archive=task/manifest['archive']
assert sha256(archive)==manifest['archive_sha256']
with zipfile.ZipFile(archive) as source:
    assert set(source.namelist())==set(manifest['files'])
    for name,digest in manifest['files'].items():
        target=destination/name
        assert target.resolve().is_relative_to(destination)
        payload=source.read(name)
        assert hashlib.sha256(payload).hexdigest()==digest
        target.parent.mkdir(parents=True,exist_ok=True)
        target.write_bytes(payload)
assert verify_index_files(destination)
assert sha256(destination/'manifest.json')==manifest['index_sha256']
print(json.dumps({'restored_files':len(manifest['files']), 'index_sha256':manifest['index_sha256'],'physical_index_verified':True}))

from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import shutil
import zipfile
import xml.etree.ElementTree as ET
from evaluation.common.release_artifacts import sha256, source_identity, write_json
from evaluation.benchmarks.freeze_human_benchmark import frozen_evidence
from evaluation.benchmarks.source_course_rows import source_course_rows
from campusai.retrieval.runtime_provenance import runtime_description

task=Path('evaluation/results/phase7-successor-source-20261004b')
corpus=Path('data/corpus/phase7_source_expansion/20261004b')
index=Path('.tmp/phase7-successor-source-20261004b/index')
previous=Path('.tmp/phase7-source-expansion-20261004/index')
original=Path('.tmp/hybrid-index')
screen=json.loads((task/'source_history_screen_connected.json').read_text(encoding='utf8'))
replay=json.loads((task/'restored_index_history_screen.json').read_text(encoding='utf8'))
assert screen['proposals']==replay['proposals'] and screen['excluded']==replay['excluded']
assert screen['history_datasets']==replay['history_datasets']
assert all(sha256(entry['path'])==entry['sha256'] for entry in screen['history_datasets'])
suite=ET.parse(task/'offline_suite.xml').find('testsuite')
assert int(suite.attrib['tests'])==813 and int(suite.attrib['skipped'])==2
assert int(suite.attrib['failures'])==int(suite.attrib['errors'])==0
cov=json.loads((task/'offline_coverage.json').read_text())['totals']['percent_covered']; assert cov>=85
preservation=[]
for parent in (original,previous):
    files={p.relative_to(parent).as_posix():sha256(p) for p in parent.rglob('*') if p.is_file()}
    assert all(sha256(index/name)==digest for name,digest in files.items() if name!='manifest.json')
    preservation.append({'parent_index':parent.as_posix(),'files':files,'all_document_files_preserved':True})
write_json(task/'index_preservation.json',{'status':'source_proposal','release_eligible':False,'parents':preservation})
evidence=frozen_evidence(index)
sources=json.loads((corpus/'source_manifest.json').read_text(encoding='utf8'))
docs={entry['doc_id'] for entry in sources['documents'] if entry['status']=='succeeded'}
facts=source_course_rows({k:v for k,v in evidence.items() if k[0] in docs}); assert len(facts)==138
raw=[]
for path in task.glob('*-records.json'):
    for row in json.loads(path.read_text(encoding='utf8'))['records']:
        actual=evidence[(row['doc_id'],row['chunk_id'])]
        assert row['content']==actual['content'] and row['page']==actual['page']
        raw.append(row)
assert len(raw)==47
write_json(task/'source_course_facts.json',{'status':'source_proposal','release_eligible':False,
    'index_sha256':sha256(index/'manifest.json'),'facts':facts,'new_raw_chunks_exact_match':len(raw)})
write_json(task/'source_table_visual_audit.json',{'status':'source_proposal','release_eligible':False,
    'recorded_at':datetime.now(timezone.utc).isoformat(),'performed_by':'Codex','independent_human_review_claim':False,
    'reviewed_previews':['telecom-page-'+str(n)+'.png' for n in (1,4,5,6,7,8,9,10)]
                        +['mechanics-page-1.png','mechanics-page-2.png'],
    'checks':['Explicit course/code/name/credit column roles checked against original PDF renders.',
              'Telecommunications: 5 supplemental rows plus 29 program rows; preserve source translations and spelling.',
              'Mechanics: 8 named coursework rows plus EMA7205 thesis, 18 credits. Uncoded elective placeholders excluded.',
              'The mechanics URL publication year 2019 is not a stated academic cohort; academic_year stays unspecified.',
              'Blank prerequisites remain unknown; shared course codes/aliases do not count as independent subjects.'],
    'limits':['Automatic history screening is not completed semantic exposure approval.',
              'No new benchmark, heldout eligibility or release PASS claim.']})
workspace=Path.cwd(); paths=set()
for directory in ('src','evaluation','scripts','tests'):
    paths.update(p for p in (workspace/directory).rglob('*.py') if p.is_file() and '__pycache__' not in p.parts)
paths.update(p for p in (workspace/'configs').rglob('*') if p.is_file() and p.suffix.lower() in {'.json','.toml','.yaml','.yml'} and p.name!='gemini.yaml')
paths.update(workspace/name for name in ('pyproject.toml','requirements.txt','requirements-dev.txt','uv.lock','poetry.lock') if (workspace/name).is_file())
identity=source_identity(workspace); snapshot=task/'final_audit_source.zip'; assert not snapshot.exists()
with zipfile.ZipFile(snapshot,'w',compression=zipfile.ZIP_DEFLATED) as archive:
    for path in sorted(paths): archive.writestr(path.relative_to(workspace).as_posix(),path.read_bytes())
with zipfile.ZipFile(snapshot) as archive:
    digest=hashlib.sha256()
    for name in sorted(archive.namelist()):
        rel=name.encode('utf8'); digest.update(len(rel).to_bytes(4,'big')); digest.update(rel)
        digest.update(hashlib.sha256(archive.read(name)).digest())
assert digest.hexdigest()==identity['source_sha256']
write_json(task/'final_audit_source_manifest.json',{**identity,'runtime':runtime_description(),
    'snapshot':snapshot.name,'snapshot_sha256':sha256(snapshot),
    'files':{p.relative_to(workspace).as_posix():sha256(p) for p in sorted(paths)},
    'meaning':'Physical source/CPU runtime for transitive family screening and offline suite; existing dirty helper retained, not a clean release. Earlier GPU build retains its original source/runtime.'})
write_json(task/'task_summary.json',{'schema_version':1,'status':'source_preparation_complete','release_eligible':False,
    'recorded_at':datetime.now(timezone.utc).isoformat(),'index_documents':14,'added_readable_documents':2,
    'added_chunks':47,'added_explicit_course_rows':43,'total_explicit_course_rows':138,
    'index_sha256':sha256(index/'manifest.json'),'proposed_connected_groups':47,
    'history_question_occurrences':screen['history_question_occurrences'],'history_datasets':len(screen['history_datasets']),
    'history_semantic_approval_complete':False,'new_benchmark_created':False,'new_heldout_or_model_scores_measured':False,
    'index_restore_and_history_screen_match':True,'old_document_bytes_preserved':True,
    'offline_tests':{'passed':811,'skipped':2,'coverage_percent':cov,'live_provider_acceptance':False},
    'official_release':{'cycle':'phase7-release-20261004-r18','passed_gates':3,'total_gates':19,'M2':'FAIL','score':None,'all_release_gates_pass':False},
    'next':'Complete semantic exposure review, validate fresh source-question schema and dev inheritance under the expanded index, preflight independent group quotas, freeze successor before heldout measurements.'})
for name in ('phase7_finalize_source_b.py','phase7_restore_index_b.py'):
    shutil.copyfile(Path('.tmp')/name,task/'reproduction'/(name+'.txt'))
print(json.dumps({'tests_passed':811,'coverage_percent':cov,'proposed_groups':47,'source_sha256':identity['source_sha256'],'history_datasets':len(screen['history_datasets'])}))

"""Reproduce the Phase 7/8 continuation review without provider requests."""
import hashlib
import json
from pathlib import Path
import subprocess
import sys
from datetime import datetime, timezone
import xml.etree.ElementTree as ET

sys.path.insert(0, str(Path(__file__).resolve().parents[3]))
from evaluation.common.release_artifacts import read_json, sha256, source_identity, write_json

ROOT = Path.cwd()
OUT = ROOT / 'evaluation/results/phase7-free-tier-performance-20261013'
RAW8 = ROOT / '.local_artifacts/experiment_runs/phase8-current-runtime-20261013'
RAW7 = ROOT / '.local_artifacts/experiment_runs/phase7-free-tier-performance-20261013'
INTEGRATION = ROOT / 'evaluation/results/phase7-live-integration-20261013'


def binding(path):
    return {'path': path.relative_to(ROOT).as_posix(), 'sha256': sha256(path)}


def test_counts(path):
    suites = ET.parse(path).getroot().findall('testsuite')
    totals = {key: sum(int(s.get(key, 0)) for s in suites)
              for key in ('tests', 'failures', 'errors', 'skipped')}
    totals['passed'] = totals['tests'] - totals['failures'] - totals['errors'] - totals['skipped']
    assert totals['errors'] == totals['failures'] == 0
    return totals


# Prove that the only source changes since the passing quality/integration runs
# are the performance dispatcher and its tests. Do not rewrite historical hashes.
old = read_json(ROOT / 'evaluation/results/phase7-chat-quality-20261013/completion-review.json')['source_identity']
changed = {'scripts/benchmarks/product_performance.py', 'tests/test_product_performance.py'}
files = set()
for directory in ('src', 'evaluation', 'scripts', 'tests'):
    files.update(p for p in (ROOT / directory).rglob('*.py') if p.is_file() and '__pycache__' not in p.parts)
files.update(p for p in (ROOT / 'configs').rglob('*') if p.is_file()
             and p.suffix.lower() in {'.json', '.toml', '.yaml', '.yml'}
             and p.name not in {'gemini.yaml', 'key.yaml'})
files.update(ROOT / n for n in ('pyproject.toml', 'requirements.txt', 'requirements-dev.txt', 'uv.lock', 'poetry.lock')
             if (ROOT / n).is_file())
reconstructed = hashlib.sha256()
deltas = []
for path in sorted(files, key=lambda p: p.relative_to(ROOT).as_posix()):
    name = path.relative_to(ROOT).as_posix()
    content = subprocess.check_output(['git', 'show', old['commit'] + ':' + name]) if name in changed else path.read_bytes()
    key = name.encode()
    reconstructed.update(len(key).to_bytes(4, 'big'))
    reconstructed.update(key)
    reconstructed.update(hashlib.sha256(content).digest())
    if name in changed:
        deltas.append({'path': name, 'before_sha256': hashlib.sha256(content).hexdigest(), 'after_sha256': sha256(path)})
assert reconstructed.hexdigest() == old['source_sha256']
current = source_identity(ROOT)
assert current['runtime_sha256'] == old['runtime_sha256']

integration = read_json(INTEGRATION / 'verified-live.json')
assert integration['status'] == 'LIVE_TWO_INTEGRATIONS_PASS'
assert integration['source_identity'] == old and integration['source_stable']
assert integration['external_provider_calls'] == 2
assert all(o['response']['modelVersion'] == 'gemini-3.8-flash' for o in integration['observations'])
live_tests = test_counts(INTEGRATION / 'verified-live.xml')
assert live_tests['passed'] == 2 and live_tests['skipped'] == 0
preflight = read_json(RAW7 / 'verified-preflight/summary.json')
assert preflight['status'] == 'OFFLINE_INTERFACE_PASS_NOT_LIVE_PERFORMANCE'
assert preflight['http_requests'] == 20 and preflight['external_provider_calls'] == 0
performance = read_json(RAW7 / 'live/summary.json')
assert performance['source_identity'] == current and performance['source_stable']
assert performance['status'] == 'FAILURE_STOPPED' and performance['external_provider_calls'] == 1
child = read_json(RAW7 / 'live/cycle-1.json')
assert sha256(RAW7 / 'live/cycle-1.json') == performance['processes'][0]['sha256']
observation = child['observations'][0]
assert observation['calls'][0]['provider_http_status'] == 503
assert observation['provider_transport_attempts'] == 1 and not observation['valid']
review8 = read_json(RAW8 / 'baseline-review.json')
assert review8['exit']['content_and_local_regression_pass']
assert read_json(RAW8 / 'execution.json')['source_identity'] == old
for path, digest in review8['bindings'].items():
    assert sha256(ROOT / path) == digest, path
tests = test_counts(RAW8 / 'final-tests.xml')
coverage = read_json(RAW8 / 'final-coverage.json')['totals']['percent_covered']
assert coverage >= 85

bound = [INTEGRATION / 'verified-egress-packet.json', INTEGRATION / 'verified-live.json',
         INTEGRATION / 'verified-live.xml', INTEGRATION / 'collector-regression-tests.xml',
         RAW8 / 'baseline-review.json', RAW8 / 'execution.json', RAW8 / 'final-tests.xml',
         RAW8 / 'final-coverage.json', RAW7 / 'verified-preflight/summary.json',
         RAW7 / 'live/summary.json', RAW7 / 'live/cycle-1.json', OUT / 'egress-packet.json',
         ROOT / 'evaluation/results/phase7-chat-quality-20261013/paired-quality-review.json', Path(__file__).resolve()]
report = {
    'schema_version': 1, 'recorded_at': datetime.now(timezone.utc).isoformat(),
    'status': 'LOCAL_AND_LIVE_INTEGRATION_PASS_PERFORMANCE_STOPPED_HTTP503',
    'reviewer': 'Codex - Automated AI Review', 'source_identity': current,
    'source_compatibility': {'prior_identity': old, 'reconstructed_prior_source_sha256': reconstructed.hexdigest(),
        'only_changed_files': deltas, 'production_source_config_runtime_unchanged': True},
    'phase7': {'done': 4, 'total': 6, 'percent': 100 * 4 / 6, 'complete': False,
        'quality': 'C7-03 PASS owner-mediated Chat; 42/42 positives, 26/26 negatives',
        'live_integrations': {'status': 'PASS', 'tests': live_tests, 'provider_posts': 2,
            'model': 'gemini-3.8-flash', 'provider_ms': [o['latency_ms'] for o in integration['observations']]},
        'performance': {'status': 'FAILURE_STOPPED_HTTP503', 'planned_http_requests': 100,
            'observed_http_requests': 1, 'valid_performance_requests': 0,
            'planned_provider_posts': 54, 'actual_provider_posts': 1, 'retry_posts': 0,
            'fallback_http_ms': observation['http_ms'], 'application_schema_valid': True,
            'provider_format': 'unmeasured: no final text', 'usage_and_cost': 'unknown: no provider usage on 503',
            'resources': child['resources'], 'resume_allowed': False, 'accepted': False},
        'next': 'C7-05 requires a complete valid live performance run after provider recovery; C7-06 depends on C7-05. No additional call or new packet scheduled.'},
    'phase8': {'done': 4, 'total': 5, 'percent': 80, 'complete': False,
        'current_application_replay_counts': review8['counts'], 'content_and_local_regression_pass': True,
        'recorded_replay_provider_posts': 0, 'fresh_validation': False,
        'historical_live_workload': {'planned': 18, 'attempted': 5, 'successful': 3, 'http503': 2, 'unattempted': 13},
        'next': 'Measure the 13 unattempted live workload cases after provider recovery. The Phase 7 synthetic integrations and different-index performance run do not fill Phase 8 cases.'},
    'offline_validation': {'full_suite': tests, 'coverage_percent': coverage,
        'collector_regression': test_counts(INTEGRATION / 'collector-regression-tests.xml')},
    'this_continuation_external_posts': 3,
    'free_tier': {'owner_declared': True, 'billing_changed': False, 'paid_fallback': False,
        'credential': 'key_2', 'key_rotation': False, 'account_billing_independently_verified': False,
        'daily_cap': 18, 'timezone': 'America/Los_Angeles'},
    'collection_improvement': 'Successful complete cycles may pause at the daily cap and resume once capacity exists in the SAME packet, without duplicates. Failure/crash/source drift remains terminal. Current run failed and is not resumable.',
    'bindings': {p.relative_to(ROOT).as_posix(): sha256(p) for p in bound},
}
write_json(OUT / 'progress-review.json', report)
print(json.dumps({'status': report['status'], 'offline_passed': tests['passed'],
                  'live_integrations_passed': live_tests['passed'], 'external_posts': 3}))

"""Fetch original public UET PDFs into a separate, non-release source proposal."""
from concurrent.futures import ThreadPoolExecutor
from urllib.request import Request, urlopen
from urllib.parse import urlparse
from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json

root = Path('.tmp/phase7-source-expansion-20261004')
root.mkdir(exist_ok=True)
sources = [
    ('uet_robotics_curriculum_2020.pdf', 'https://uet.vnu.edu.vn/wp-content/uploads/2018/04/KT-Robot-150TC.pdf'),
    ('uet_engineering_physics_curriculum_2019.pdf', 'https://fepn.uet.vnu.edu.vn/upload/2001767/fck/files/f345c47deadba3beeb5f8e4e923fc042%281%29.pdf'),
]
def fetch(spec):
    name, url = spec
    path = root/name
    assert not path.exists(), 'preserve prior downloads'
    try:
        with urlopen(Request(url, headers={'User-Agent':'CampusAI corpus audit/1.0'}), timeout=25) as response:
            assert urlparse(response.url).hostname.endswith('vnu.edu.vn')
            payload = response.read(20000001)
            assert payload.startswith(b'%PDF-') and len(payload) <= 20000000
            final_url = response.url
        with path.open('xb') as stream:
            stream.write(payload)
        return {'filename':name,'source_url':url,'final_url':final_url,'status':'downloaded',
                'bytes':len(payload),'sha256':hashlib.sha256(payload).hexdigest()}
    except Exception as error:
        return {'filename':name,'source_url':url,'status':'fetch_failed','error':str(error)}
with ThreadPoolExecutor(max_workers=2) as executor:
    result = list(executor.map(fetch, sources))
manifest = {'status':'source_proposal', 'release_eligible':False,
            'downloaded_at':datetime.now(timezone.utc).isoformat(),'documents':result}
(root/'download_receipt.json').write_text(json.dumps(manifest,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
print(json.dumps(manifest,ensure_ascii=False,indent=2))

"""Audit original PDF ingestion and unlabelled source-fact history candidates."""
from pathlib import Path
from collections import Counter
from datetime import datetime, timezone
import json
import re
import pymupdf
from campusai.ingestion.pipeline import ingest_document
from campusai.retrieval.index_builder import chunk_records
from evaluation.benchmarks.source_course_rows import source_course_rows, normalized_header
from evaluation.benchmarks.connected_groups import connected_groups
from evaluation.common.release_artifacts import sha256, write_json

root=Path('.tmp/phase7-source-expansion-20261004')
receipts=[json.loads((root/name).read_text(encoding='utf-8')) for name in ('download_receipt.json','additional_download_receipt.json')]
source_entries=[entry for receipt in receipts for entry in receipt['documents'] if entry['status']=='downloaded']
all_facts=[]
documents=[]
for entry in source_entries:
    path=root/entry['filename']
    assert sha256(path)==entry['sha256']
    with pymupdf.open(path) as pdf:
        page_text=[page.get_text() for page in pdf]
        count=len(pdf)
        for number in [0]+[page.number for page in pdf if any(label in page.get_text() for label in ('Số tín','tín chỉ','tín\nchỉ'))]:
            pdf[number].get_pixmap(matrix=pymupdf.Matrix(1.3,1.3)).save(root/f"{path.stem}-page-{number+1}.png")
    document=ingest_document(path,store_dir=root/'store',enable_ocr=False)
    # Metadata is input corpus context; source row roles must still be explicit.
    year=re.search(r'20\d{2}',path.stem).group(0)
    document.metadata.update(document_type='curriculum',language='vi-en',academic_year=year,
                             institution='VNU-UET',source_sha256=entry['sha256'],source_url=entry['source_url'])
    for chunk in document.chunks:
        chunk.metadata.update(document.metadata)
    records=chunk_records(document)
    write_json(root/(path.stem+'-records.json'),{'records':records})
    facts=source_course_rows({(r['doc_id'],r['chunk_id']):r for r in records})
    all_facts.extend(facts)
    documents.append({**entry,'doc_id':document.doc_id,'pages':count,'status':document.status,
                      'chunks':len(records),'tables':sum(r['content_type']=='table' for r in records),
                      'explicit_course_rows':len(facts),'text_characters_by_page':[len(t) for t in page_text]})

history=[]
datasets=[]
for directory in (Path('data/benchmark'),Path('evaluation/results/phase7-dev-feasibility-20261004/exposed_dev_probe')):
    for path in sorted(directory.rglob('*.jsonl')):
        if 'owner_confirmed_inputs' in path.parts:
            continue
        datasets.append({'path':path.as_posix(),'sha256':sha256(path)})
        for line in path.read_text(encoding='utf-8-sig').splitlines():
            if line.strip():
                row=json.loads(line)
                question=row.get('question',row.get('query',''))
                if question:
                    history.append(normalized_header(question))
history_text='\n'.join(history)
compact_history=re.sub(r'\s+','',history_text)
fresh=[]
excluded=[]
for fact in all_facts:
    code_match=re.search(r'(?<![a-z0-9])'+fact['code'].casefold()+r'(?![a-z0-9])',history_text)
    name=normalized_header(fact['name'])
    # This is a conservative automatic screen, not a semantic exposure approval.
    lines = [normalized_header(line) for line in fact['name_lines']]
    names = [name] + [' '.join(lines[start:end])
                     for start in range(len(lines))
                     for end in range(start + 1, len(lines) + 1)]
    overlap=next((alias for alias in names if len(alias)>=8 and re.sub(r'\s+','',alias) in compact_history),None)
    if code_match or overlap:
        excluded.append({'fact':fact,'reason':'historical_code_or_name_mention','matched_name':overlap})
    else:
        fresh.append(fact)
groups=connected_groups([{'qid':str(number),'question':'','paraphrase_group':fact['code'],
                          'source_group':normalized_header(fact['name'])}
                         for number,fact in enumerate(fresh)])
for number,fact in enumerate(fresh):
    fact['connected_group']=groups[str(number)]
report={'schema_version':1,'status':'source_proposal','release_eligible':False,
        'recorded_at':datetime.now(timezone.utc).isoformat(),'documents':documents,'history_datasets':datasets,
        'history_question_occurrences':len(history),'explicit_course_rows':len(all_facts),
        'proposed_rows':len(fresh),'proposed_connected_groups':len(set(groups.values())),
        'proposals_by_document':dict(Counter(f['doc_id'] for f in fresh)),
        'excluded':excluded,'proposals':fresh,
        'limits':['No retrieval/model scoring on proposed facts/questions.','No new benchmark records, approvals or heldout eligibility declared.',
                  'Exact source headers/row hashes support values; PDF table rendering and semantic exposure review still required before benchmark freeze.']}
write_json(root/'source_fact_proposal_audit.json',report)
print(json.dumps({k:report[k] for k in ('documents','explicit_course_rows','proposed_rows','proposed_connected_groups')},ensure_ascii=False,indent=2))

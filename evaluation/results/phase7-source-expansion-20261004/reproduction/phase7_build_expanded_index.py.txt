"""Build a separate real BGE-M3 index, preserving every original index byte."""
from pathlib import Path
import json
import shutil
from datetime import datetime, timezone
from campusai.ingestion.pipeline import ingest_document
from campusai.retrieval.index_builder import build_document_indexes
from campusai.retrieval.model_runtime import retrieval_runtime
from evaluation.common.release_artifacts import sha256, verify_index_files, write_json, source_identity

root=Path('.tmp/phase7-source-expansion-20261004')
old=Path('.tmp/hybrid-index')
index=root/'index'
assert not index.exists(), 'preserve prior index builds'
old_manifest=json.loads((old/'manifest.json').read_text(encoding='utf-8'))
old_hash=sha256(old/'manifest.json')
old_bytes={path.relative_to(old).as_posix():sha256(path) for path in old.rglob('*') if path.is_file()}
before=source_identity(Path.cwd())
shutil.copytree(old,index)
audit=json.loads((root/'source_fact_proposal_audit.json').read_text(encoding='utf-8'))
added=[]
for entry in audit['documents']:
    if entry['status']!='succeeded' or entry['chunks']==0:
        continue
    path=root/entry['filename']
    assert sha256(path)==entry['sha256']
    document=ingest_document(path,store_dir=root/'store',enable_ocr=False)
    document.metadata.update(document_type='curriculum',language='vi-en',
        academic_year=entry['filename'].rsplit('_',1)[-1].split('.')[0],institution='VNU-UET',
        source_sha256=entry['sha256'],source_url=entry['source_url'])
    for chunk in document.chunks:
        chunk.metadata.update(document.metadata)
    build_document_indexes(document,index,dense_model='BAAI/bge-m3',device='cuda',batch_size=8)
    added.append(document.doc_id)
assert sha256(old/'manifest.json')==old_hash and all(sha256(old/name)==digest for name,digest in old_bytes.items())
assert all(sha256(index/name)==digest for name,digest in old_bytes.items() if name!='manifest.json')
assert verify_index_files(index)
manifest=json.loads((index/'manifest.json').read_text(encoding='utf-8'))
assert set(manifest['documents'])==set(old_manifest['documents'])|set(added)
encoder=retrieval_runtime().get_encoder('BAAI/bge-m3','cuda')
revision=getattr(encoder._first_module().auto_model.config,'_commit_hash',None)
after=source_identity(Path.cwd())
assert before==after
report={'schema_version':1,'status':'source_proposal','release_eligible':False,
        'built_at':datetime.now(timezone.utc).isoformat(),'index_path':index.as_posix(),
        'old_index_sha256':old_hash,'index_sha256':sha256(index/'manifest.json'),
        'original_index_unchanged':True,'original_document_bytes_preserved':True,
        'documents':len(manifest['documents']),'added_documents':added,
        'physical_index_verified':True,'model_name':'BAAI/bge-m3','physical_model_revision':revision,
        'device':'cuda','batch_size':8,'embedding_dimension':manifest['embedding_dimension'],
        'source_before':before,'source_after':after,'source_stable':True,
        'next':'Bind source facts to this physical index; audit exposure and freeze a successor benchmark before any heldout measurement.'}
write_json(root/'expanded_index_build_receipt.json',report)
print(json.dumps(report,indent=2))

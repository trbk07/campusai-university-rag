from pathlib import Path
import hashlib
import zipfile
from evaluation.common.release_artifacts import source_identity, sha256, write_json
from campusai.retrieval.runtime_provenance import runtime_description

workspace = Path.cwd()
root = workspace/'.tmp/phase7-source-expansion-20261004'
receipt = __import__('json').loads((root/'expanded_index_build_receipt.json').read_text(encoding='utf-8'))
identity = source_identity(workspace)
assert identity == receipt['source_before']
files = set()
for directory in ('src','evaluation','scripts','tests'):
    files.update(p for p in (workspace/directory).rglob('*.py') if p.is_file() and '__pycache__' not in p.parts)
files.update(p for p in (workspace/'configs').rglob('*') if p.is_file() and p.suffix.lower() in {'.json','.toml','.yaml','.yml'} and p.name != 'gemini.yaml')
files.update(workspace/n for n in ('pyproject.toml','requirements.txt','requirements-dev.txt','uv.lock','poetry.lock') if (workspace/n).is_file())
snapshot = root/'index_build_source.zip'
assert not snapshot.exists()
with zipfile.ZipFile(snapshot,'w',compression=zipfile.ZIP_DEFLATED) as archive:
    for path in sorted(files):
        archive.writestr(path.relative_to(workspace).as_posix(),path.read_bytes())
with zipfile.ZipFile(snapshot) as archive:
    digest=hashlib.sha256()
    for name in sorted(archive.namelist()):
        rel=name.encode('utf-8')
        digest.update(len(rel).to_bytes(4,'big')); digest.update(rel)
        digest.update(hashlib.sha256(archive.read(name)).digest())
assert digest.hexdigest()==identity['source_sha256']
write_json(root/'index_build_source_manifest.json',{**identity,'snapshot':snapshot.name,'snapshot_sha256':sha256(snapshot),'runtime':runtime_description(),
    'files':{p.relative_to(workspace).as_posix():sha256(p) for p in sorted(files)},
    'meaning':'Exact historical worktree during GPU indexing, including pre-existing dirty source helpers; conditional source preparation, not clean release acceptance. Credentials excluded.'})
print({'snapshot_files':len(files),'source_sha256':identity['source_sha256']})

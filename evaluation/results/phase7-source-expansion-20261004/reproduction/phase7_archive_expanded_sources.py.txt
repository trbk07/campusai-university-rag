from pathlib import Path
from datetime import datetime, timezone
import gzip
import hashlib
import json
import shutil
import zipfile
from evaluation.common.release_artifacts import sha256, write_json
from evaluation.benchmarks.freeze_human_benchmark import frozen_evidence
from evaluation.benchmarks.source_course_rows import source_course_rows

local=Path('.tmp/phase7-source-expansion-20261004')
corpus=Path('data/corpus/phase7_source_expansion/20261004')
task=Path('evaluation/results/phase7-source-expansion-20261004')
assert not corpus.exists() and not task.exists()
corpus.mkdir(parents=True); task.mkdir(parents=True)
(corpus/'.gitignore').write_text('!*.pdf\n',encoding='utf8')
for folder in (corpus,task):
    (folder/'.gitattributes').write_text('* -text\n* whitespace=cr-at-eol\n',encoding='utf8')
audit=json.loads((local/'source_fact_proposal_audit.json').read_text(encoding='utf8'))
for entry in audit['documents']:
    shutil.copyfile(local/entry['filename'],corpus/entry['filename'])
    assert sha256(corpus/entry['filename'])==entry['sha256']
write_json(corpus/'source_manifest.json',{'schema_version':1,'status':'source_proposal','release_eligible':False,
    'documents':audit['documents'],
    'meaning':'Original public university PDFs, retaining document years, not a claim that these curricula are current in 2026. Only two text-readable documents are indexed; scan-only Nano biology document is excluded.',
    'download_receipts':['download_receipt.json','additional_download_receipt.json']})
for name in ('download_receipt.json','additional_download_receipt.json'):
    shutil.copyfile(local/name,corpus/name)
for name in ('source_fact_proposal_audit.json','expanded_index_build_receipt.json','index_build_source.zip','index_build_source_manifest.json'):
    shutil.copyfile(local/name,task/name)
reproduction=task/'reproduction'; reproduction.mkdir()
for name in ('phase7_fetch_successor_sources.py','phase7_fetch_more_successor_sources.py',
             'phase7_audit_expanded_source_facts.py','phase7_build_expanded_index.py','phase7_snapshot_expanded_build.py',
             'phase7_archive_expanded_sources.py'):
    shutil.copyfile(Path('.tmp')/name,reproduction/(name+'.txt'))
previews=task/'source_previews'; previews.mkdir()
preview_names=[f'uet_robotics_curriculum_2020-page-{n}.png' for n in (1,11,12,13,14,15,16)]
preview_names += [f'uet_electronics_masters_curriculum_2022-page-{n}.png' for n in (1,6,7,8,9)]
for name in preview_names:
    shutil.copyfile(local/name,previews/name)
raw_archives=[]
for path in sorted(local.glob('*-records.json')):
    archive=task/(path.name+'.gz')
    with path.open('rb') as src,gzip.GzipFile(filename=str(archive),mode='wb',mtime=0) as dest:
        shutil.copyfileobj(src,dest)
    with gzip.open(archive,'rb') as stream:
        restored=hashlib.file_digest(stream,'sha256').hexdigest()
    assert restored==sha256(path)
    raw_archives.append({'archive':archive.name,'sha256':sha256(archive),'restored_sha256':restored,'bytes':path.stat().st_size})
evidence=frozen_evidence(local/'index')
facts=source_course_rows({k:v for k,v in evidence.items() if k[0] in {e['doc_id'] for e in audit['documents'] if e['status']=='succeeded'}})
raw=[]
for entry in audit['documents']:
    raw.extend(json.loads((local/(Path(entry['filename']).stem+'-records.json')).read_text(encoding='utf8'))['records'])
for row in raw:
    actual=evidence[(row['doc_id'],row['chunk_id'])]
    assert actual['content']==row['content'] and actual['page']==row['page']
assert len(facts)==95
archive=task/'expanded_index.zip'
files={path.relative_to(local/'index').as_posix():sha256(path) for path in (local/'index').rglob('*') if path.is_file()}
with zipfile.ZipFile(archive,'w',compression=zipfile.ZIP_DEFLATED) as target:
    for name in sorted(files):
        target.writestr(name,(local/'index'/name).read_bytes())
with zipfile.ZipFile(archive) as target:
    assert set(target.namelist())==set(files)
    assert all(hashlib.sha256(target.read(name)).hexdigest()==digest for name,digest in files.items())
write_json(task/'index_archive_manifest.json',{'status':'source_proposal','release_eligible':False,
    'archive':archive.name,'archive_sha256':sha256(archive),'files':files,
    'index_sha256':sha256(local/'index/manifest.json'),'restored_files_verified':len(files),
    'all_added_chunks_match_ingestion':len(raw),'explicit_course_rows_bound_to_index':len(facts),
    'physical_model_revision':'5617a9f61b028005a4858fdac845db406aefb181',
    'model_revision_field_meaning':'Existing index schema serializes provider version 1; the actual cached Hugging Face commit is independently recorded in the GPU build receipt. Preserve original artifact bytes; do not reinterpret 1 as a Hugging Face commit.'})
write_json(task/'raw_archives.json',{'status':'source_proposal','release_eligible':False,'archives':raw_archives})
write_json(task/'source_table_visual_audit.json',{'status':'source_proposal','release_eligible':False,
    'recorded_at':datetime.now(timezone.utc).isoformat(),'performed_by':'Codex','independent_human_review_claim':False,
    'reviewed_previews':preview_names,'checks':['Explicit course/code/name/credit columns corroborated against PDF table renderings.',
    'Bilingual wrapped names and sparse column layouts are preserved; credits are not read from contact-hour columns.',
    'Merged credit cells in electronics page 9 carry 3 for the listed elective rows, consistent with ingestion serialization.',
    'Blank prerequisite cells are not evidence of no prerequisite.'],
    'limits':['Robot page 13 table fragmentation drops some original course rows from table serialization; only the 61 actually extracted rows are proposed.',
    'Nano biology is scan-only with no indexed chunks; no facts proposed.',
    'Physics PDF TLS hostname mismatch was retained as a download failure; TLS checks were not disabled.',
    'Semantic history audit and owner-directed benchmark creation are still pending; no heldout eligibility or PASS release claim.']})
print(json.dumps({'source_pdfs':len(audit['documents']),'index_archive_bytes':archive.stat().st_size,'index_files':len(files),'bound_facts':len(facts),'added_chunks':len(raw)}))

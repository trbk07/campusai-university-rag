historical_sources = {}


def verified_bindings(report, historical=False):
    for name, digest in report['bindings'].items():
        path = (ROOT / name).resolve()
        if not path.is_relative_to(ROOT):
            raise ValueError('historical binding mismatch: ' + name)
        if sha(path) != digest:
            # Source can evolve; preserve and verify the exact Git-published
            # source used by the historical receipt rather than rewriting it.
            if not historical or name not in {
                    'evaluation/phase8/measure_live_runtime.py', 'tests/test_phase8_live_runtime.py'}:
                raise ValueError('historical binding mismatch: ' + name)
            content = subprocess.check_output(['git', 'show', '2ac714d:' + name], cwd=ROOT)
            if hashlib.sha256(content).hexdigest() != digest:
                raise ValueError('published source differs from historical binding: ' + name)
            snapshot = live.OUTPUT / (Path(name).stem + '.snapshot.py.txt')
            if snapshot.exists() and snapshot.read_bytes() != content:
                raise ValueError('historical snapshot already differs')
            snapshot.write_bytes(content)
            historical_sources[name] = {'snapshot': snapshot.relative_to(ROOT).as_posix(),
                                        'sha256': digest, 'commit': '2ac714d'}
    return len(report['bindings'])

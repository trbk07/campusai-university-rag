import json
from pathlib import Path
from evaluation.common.release_artifacts import sha256

base = Path('evaluation/results/phase7-source-context-20261004-r18')
dataset = Path('data/benchmark/phase7_review_cycles/phase7-release-20261004-r18/human_retrieval_dev.jsonl')
report = json.loads((base / 'candidate_coverage.json').read_text(encoding='utf-8'))
part = report['splits']['human_dev']
assert part['benchmark_sha256'] == sha256(dataset)
metrics = part['caps']['40']
rows = {row['qid']: row for row in map(json.loads, dataset.read_text(encoding='utf-8').splitlines())}
misses = metrics['candidate_misses']
output = {key: metrics[key] for key in ('records', 'answerable', 'negative', 'candidate_recall', 'all_gold_recall', 'evidence_recall')}
output.update(status='diagnostic', release_eligible=False, split='human_dev',
              dataset_sha256=sha256(dataset), candidate_report_sha256=sha256(base / 'candidate_coverage.json'),
              strata=metrics['strata'], misses=[{**miss, 'gold_evidence': rows[miss['qid']]['gold_evidence'],
                                               'scope': rows[miss['qid']].get('doc_ids')} for miss in misses])
target = base / 'dev_coverage_diagnostic.json'
target.write_text(json.dumps(output, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
print(json.dumps({**{key: output[key] for key in ('records', 'answerable', 'negative', 'candidate_recall', 'all_gold_recall')},
                  'misses': [{key: miss[key] for key in ('qid', 'query', 'expected_evidence', 'gold_evidence', 'scope')} for miss in output['misses']]}, ensure_ascii=False, indent=2))

from pathlib import Path
import hashlib
import json
import subprocess

task=Path('evaluation/results/phase7-dev-inheritance-20261004')
manifest=json.loads((task/'phase7_dev_inheritance.json').read_text(encoding='utf8'))
checked=[]; missing=[]; mismatched=[]
for parent in manifest['parents']:
    for name,digest in parent['benchmark_files'].items():
        path=parent['benchmark_dir']+'/'+name
        proc=subprocess.run(['git','show','HEAD:'+path],capture_output=True)
        if proc.returncode:
            missing.append(path)
        elif hashlib.sha256(proc.stdout).hexdigest()!=digest:
            mismatched.append(path)
        else:
            checked.append(path)
print(json.dumps({'git_exact_parent_files':len(checked),'missing':missing,'mismatched':mismatched},indent=2))

from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import shutil
import subprocess
import zipfile
import xml.etree.ElementTree as ET
from evaluation.common.release_artifacts import read_json, sha256, source_identity, write_json
from campusai.retrieval.runtime_provenance import runtime_description

workspace=Path.cwd(); task=Path('evaluation/results/phase7-dev-inheritance-20261004')
preflight=read_json(task/'preflight.json'); replay=read_json(task/'loader_integration-final.json')
assert preflight['records']==replay['records_reconstructed_exactly']==replay['heldout_reassignments_rejected']==817
assert not replay['review_audit_errors']
identity=source_identity(workspace)
assert identity==replay['source_before']==replay['source_after']
manifest=read_json(task/'phase7_dev_inheritance.json')
for parent in manifest['parents']:
    for name,digest in parent['benchmark_files'].items():
        assert sha256(Path(parent['benchmark_dir'])/name)==digest
    for name,digest in parent['index_files'].items():
        assert sha256(Path(parent['index_dir'])/name)==digest
restored=Path('.tmp/phase7-inherited-parent-index-restored-20261004')
assert all(sha256(restored/name)==digest for name,digest in manifest['parents'][0]['index_files'].items())
write_json(task/'restored_parent_index_verification.json',{'status':'validated_dev_inheritance','release_eligible':False,
    'restored_files':len(manifest['parents'][0]['index_files']),'all_file_hashes_match':True,
    'index_sha256':sha256(restored/'manifest.json')})
source_files=set()
for directory in ('src','evaluation','scripts','tests'):
    source_files.update(p for p in (workspace/directory).rglob('*.py') if p.is_file() and '__pycache__' not in p.parts)
source_files.update(p for p in (workspace/'configs').rglob('*') if p.is_file() and p.suffix.lower() in {'.json','.toml','.yaml','.yml'} and p.name!='gemini.yaml')
source_files.update(workspace/name for name in ('pyproject.toml','requirements.txt','requirements-dev.txt','uv.lock','poetry.lock') if (workspace/name).is_file())
snapshot=task/'final_source.zip'; assert not snapshot.exists()
with zipfile.ZipFile(snapshot,'w',compression=zipfile.ZIP_DEFLATED) as archive:
    for path in sorted(source_files):
        archive.writestr(path.relative_to(workspace).as_posix(),path.read_bytes())
with zipfile.ZipFile(snapshot) as archive:
    digest=hashlib.sha256()
    for name in sorted(archive.namelist()):
        rel=name.encode('utf8'); digest.update(len(rel).to_bytes(4,'big')); digest.update(rel)
        digest.update(hashlib.sha256(archive.read(name)).digest())
assert digest.hexdigest()==identity['source_sha256']
write_json(task/'final_source_manifest.json',{**identity,'runtime':runtime_description(),'snapshot':snapshot.name,
    'snapshot_sha256':sha256(snapshot),'files':{p.relative_to(workspace).as_posix():sha256(p) for p in sorted(source_files)},
    'meaning':'Exact source/CPU runtime of final loader integration and offline regression. Includes pre-existing uncommitted source helper; not a clean release source. Historical preliminary integration receipt has its own earlier source hash and is not rebound.'})
reproduction=task/'reproduction'
for name in ('phase7_validate_dev_inheritance_integration.py','phase7_audit_inheritance_parent_git_normalization.py',
             'phase7_inheritance_check_git_parents.py','phase7_finalize_dev_inheritance_task.py'):
    shutil.copyfile(Path('.tmp')/name,reproduction/(name+'.txt'))
suite=ET.parse(task/'offline_suite_git_byte_guard.xml').find('testsuite')
assert int(suite.attrib['tests'])==812 and int(suite.attrib['skipped'])==2
assert int(suite.attrib['errors'])==int(suite.attrib['failures'])==0
coverage=read_json(task/'offline_coverage_git_byte_guard.json')['totals']['percent_covered']
readiness=read_json(task/'active_r18_validator_recheck.json')
assert readiness['status']=='conditional' and readiness['score'] is None and readiness['all_release_gates_pass'] is False
assert readiness['gates']['M2']['status']=='FAIL'
assert readiness['gates']['M0']['status']==readiness['gates']['M1']['status']==readiness['supplemental_gates']['S1']['status']=='PASS'
write_json(task/'task_summary.json',{'schema_version':1,'status':'dev_inheritance_preparation_complete','release_eligible':False,
    'recorded_at':datetime.now(timezone.utc).isoformat(),'records':817,'human':180,'ai':637,'historical_authorizations':3,
    'dev_only':True,'heldout_reassignments_rejected':817,'review_audit_errors':0,'parent_artifacts_bound':54,
    'original_index_files_restored':22,'git_newline_serialization_repairs':16,'physical_parent_records_changed':False,
    'new_benchmark_frozen':False,'new_questions_or_model_scores_generated':False,
    'offline_tests':{'passed':810,'skipped':2,'coverage_percent':coverage,'live_provider_acceptance':False},
    'active_release_recheck':{'status':'conditional','score':None,'all_release_gates_pass':False,'M0':'PASS','M1':'PASS','S1':'PASS','M2':'FAIL'},
    'next':'Audit enough fresh independent source groups for novel dev and both heldout splits; integrate exact inherited provenance; freeze successor and measure all gates in order.'})
checks={p.relative_to(task).as_posix():sha256(p) for p in sorted(task.rglob('*')) if p.is_file() and p.name!='artifact_checksums.json'}
write_json(task/'artifact_checksums.json',{'schema_version':1,'status':'validated_dev_inheritance','release_eligible':False,'files':checks})
print(json.dumps({'archived_files':len(checks),'source_members':len(source_files),'source_sha256':identity['source_sha256'],'records_verified':817,'tests_passed':810}))
